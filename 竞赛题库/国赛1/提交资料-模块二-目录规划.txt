打包命令（在Python 3.6 venv中执行）：
cd 2-6_门闸环境监控
D:\源码\2-6_监控管理系统\.venv\Scripts\pyinstaller.exe --onefile --name 门闸环境监控系统 门闸环境监控.py
copy dist\门闸环境监控系统.exe ..\..\exe\

cd ..\2-7_商品管理
D:\源码\2-6_监控管理系统\.venv\Scripts\pyinstaller.exe --onefile --name 商品管理系统 商品管理.py
